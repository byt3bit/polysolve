In the polynomial system $\mathcal{F}_{\mathrm{LWE}}$ (stemming from \slwe), derived from $m = \binom{n+d-1}{d}$ samples, we identify each polynomial with its coefficient vector. The terms of each polynomial are ordered according to a fixed \emph{graded} monomial ordering by total degree. 
% in the vector space of multivariate polynomials in $n$ variables of (total) degree at most $d$, which has dimension $w = \binom{n+d}{d}$. Fixing a graded monomial ordering induces a canonicalordering by total degree.
This yields a matrix $M \in \Z_q^{m \times w}$ whose rows correspond to the input polynomials and $w = \binom{n+d}{d}$.
With high probability, these rows are linearly independent (see \Cref{pivots}) and the leading principal $m \times m$ submatrix of $M$ is invertible; diagonalizing it produces a set of polynomials with monic leading terms of degree $d$ (outlined by the first square structure in \cref{fig:bigmatrix}).
The remaining columns form a dense block containing lower-degree monomials. Each polynomial in this representation has a unique term of degree $d$ corresponding to the leading term, and additional terms of lower degree. The leading terms are all distinct and, due to the linear independence of the rows, they cover all monomials of degree $d$.

% This structure can be represented as a rectangular coefficient matrix with a block decomposition: the left part, corresponding to a square block, is diagonalized and highlighted in Figure~\ref{fig:bigmatrix}, while the remaining columns form a dense block containing lower-degree monomials. 
% Each polynomial in this representation has a unique term of degree $d$ corresponding to the leading term, and additional terms of lower degree. 
% The leading terms are all distinct and, due to the linear independence of the rows, they cover all monomials of degree $d$.

From this system, we generate a selected set of $S$-polynomials each having total degree $ \leq d$. Next we reduce this set of $S$-polynomials by the current diagonalized system. Note that this process \emph{does not require division}, as it consists of linear combinations of the diagonalized
polynomials. Each reduced $S$-polynomial has total degree exactly $d-1$.

The coefficients of these reduced $S$-polynomials are then assembled into a new coefficient matrix.
Provided that a sufficient number of such polynomials is generated, the leading principal submatrix of this matrix is invertible with overwhelming probability and can be diagonalized in the same manner as before.
This yields a new diagonalized system whose leading terms are monic and cover all monomials of degree $d-1$. Then we compute a set of $S$-polynomials and their reduction as described before. 

This procedure is applied recursively, decreasing the degree at each iteration. Finally, the process terminates at degree $1$, where the resulting system consists of linear polynomials of the form $(x_j - s_j)$ for $j=1,\dots,n$, and directly reveal the secret vector $\vars{s}$.

\paragraph*{Iterations (formal)}
The main algorithm (\cref{alg:alg}) has $d-1$ iterations, processing polynomials of total degree $i+1$ at iteration-$i$, starting with $i = d-1$ and decreasing by one at each iteration until $i=1$. 
At iteration-$i$, it performs the
following operations:  

% \begin{definition}[$\deg_i$ iteration]
% For $1 \le i \le d-1$, the \emph{$\deg_i$ iteration} refers to the stage in which polynomials with leading terms of degree $i$ are processed. Specifically, $S$-polynomials of degree $i$ are generated and subsequently diagonalized.
% \end{definition}

% With this notation, the $\deg_i$ iteration corresponds to the $(d-i)$-th step of the algorithm. Each $\deg_i$ iteration consists of two main operations:

\begin{enumerate}
    \item \textbf{$S$-polynomial generation.} (\texttt{Spoly})
    We generate $\binom{n+i-1}{i}$ $S$-polynomials, ensuring that the set contains exactly as many equations as there are monomials of degree $i$.

    An $S$-polynomial between polynomials $f$ and $g$ is computed if the greatest common divisor of their (distinct) leading terms has degree $i$.
    We recall that at this stage, \LT$(f)$ and \LT$(g)$ have degree $i+1$.
    The resulting $S$-polynomial has the form  
    \begin{equation}\label{eq:spoly}
        S_{fg} = x_k\cdot f - x_\ell \cdot g
    \end{equation}
    for suitable indices $k,\ell \in \{1,\dots,n\}$.
    As a consequence, $S_{fg}$ consists only of monomials of degree at most $i+1$.
    After reduction with respect to the starting set of polynomials, all terms in $S_{fg}$ have degree at most $i$.

    \item \textbf{Diagonalization.} (\texttt{Diag}) \mlf{All in one step.}
    The leading principal submatrix of size $\binom{n+i-1}{i}$ is diagonalized in two stages: 
    \begin{enumerate}
        \item (\texttt{DiagA}) compute a row-echelon form to obtain an upper triangular matrix by eliminating entries below the diagonal;
        \item (\texttt{DiagB}) complete the diagonalization by eliminating entries above the diagonal. 
    \end{enumerate}
  \end{enumerate}

  \begin{algorithm}[t!]
\caption{Main Algorithm}\label{alg:alg}
\begin{algorithmic}[1]
\Statex{\hspace*{-\algorithmicindent} \textbf{Input:} $\mathcal{P}_d$: initial set of samples, $n \geq 1$: secret's dimension, $d$: maximum error size}
\Statex{\hspace*{-\algorithmicindent} \textbf{Output:} $\mathcal{D}_1$: set of $n$ linear polynomials $\{ x_j - s_j: 1 \le j \le n \}$.}
% \Statex{}
\State{\texttt{SET = DiagB(DiagA($\mathcal{P}_d, n, d$))}} \Comment{Initial diagonalization of the degree-$d$ samples}
\For{$i = d-1, d-2, \dots, 1$}
    \State{\texttt{STEP1 = Spoly(SET,$n, i$)}} \Comment{$S$-polynomial generation step: $\mathcal{P}_i$}
    \State{\texttt{SET = DiagB(DiagA(STEP1,$n, i$))}} \Comment{Diagonalization step: $\mathcal{D}_i$}
\EndFor{}
\State{\Return{$\mathcal{D}_1$}} \Comment{Return the reduced set of linear polynomials}
\end{algorithmic}
\end{algorithm}
