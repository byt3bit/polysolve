% STEP 1 - PROOF of the WHOLE STEP
\begin{proof}[Proof of \Cref{thm:iterationCost} (i)]
\textbf{Complexity of \texttt{Spoly}.}
We prove overall complexity of \texttt{Spoly}, $\mathscr{C}_\texttt{Spoly}$, by first analyzing the complexity of a single call, and then summing over all $d-1$ calls. 
Let $1 \le i \le d-1$ and consider the diagonalized set $\mathcal{D}_{i+1}$.

% STEP 1 - DESCRIPTION of 1 ITERATION
\underline{\textsf{Claim 1.}}
% \begin{claim}
    The cost of generating and reducing $\mathcal{P}_i$ from $\mathcal{D}_{i+1}$ at the iteration-$i$ is
    \[
        2\cdot \binom{n+i-1}{i}^2\cdot \left[ \binom{n+i}{i} + 1 \right].
    \]
% \end{claim}
% \begin{claimproof}
\underline{\textsf{Proof of Claim 1.}}
Consider the iteration-$i$, with $1\le i \le d-1$, and let $\mathcal{D}_{i+1}$ be the current diagonalized set.
We have
\[
|\mathcal{D}_{i+1}| = \binom{n+i}{i+1}, \qquad
|\mathcal{P}_i| = \binom{n+i-1}{i}.
\]

Each polynomial in $\mathcal{D}_{i+1}$ contains exactly one monomial of degree $i+1$ and at most $\binom{n+i-1}{i}$ monomials of degree $i$, namely those monomials which, when multiplied by a variable, yield a monomial of degree $i+1$.
Consequently, in the $S$-polynomial $S_{fg}$, at most $2 \cdot \binom{n+i-1}{i}$ monomials of degree $i+1$ arise, where the factor $2$ accounts for the contributions coming from both $f$ and $g$.

Each of these degree-$(i+1)$ monomials is reduced using exactly one polynomial from $\mathcal{D}_{i+1}$, thanks to its diagonal structure.
Such a polynomial contains at most $\binom{n+i}{i}+1$ monomials.
Hence, the cost (in scalar multiplications) to reduce a single $S$-polynomial is
%\[
$2 \cdot \binom{n+i-1}{i} \cdot \left[\binom{n+i}{i}+1\right]$.
%\]
Since $\binom{n+i-1}{i}$ $S$-polynomials are generated at the iteration-$i$, the total cost is
\[
\sum_{j=1}^{\binom{n+i-1}{i}} \left[ 2 \cdot \binom{n+i-1}{i} \cdot \left[ \binom{n+i}{i} + 1 \right] \right]
=
2 \cdot \binom{n+i-1}{i}^2 \cdot \left[ \binom{n+i}{i} + 1 \right],
\]
proving the {\textsf{Claim 1.}}
% \end{claimproof}

% CONTINUATION OF MAIN PROOF
The total complexity of the $S$-polynomial generation and reduction is obtained by summing the contribution of the $d-1$ iterations over all degrees $1\leq i \leq d-1$.
Using the following relation $\binom{n+i}{i} =\frac{n+i}{n}\binom{n+i-1}{i}$, we have complexity $\mathscr{C}_{\texttt{Spoly}}$ as
    \begin{align*}
        \mathscr{C}_{\texttt{Spoly}}
        &:=\sum_{i=1}^{d-1} \left[  2 \cdot \binom{n+i-1}{i}^2 \cdot \left[ \binom{n+i}{i} + 1 \right] \right]\\
      % &= \sum_{i=1}^{d-1} \Bigg[ 2 \binom{n+i-1}{i}^2  \binom{n+i}{i} + 2\binom{n+i-1}{i}^2  \Bigg]
        &= \sum_{i=1}^{d} \left[ 2\frac{n+i}{n}\binom{n+i-1}{i}^3 + 2\binom{n+i-1}{i}^2  \right],
    \end{align*}
which gives for $i=d-1$ the highest term.
Observing that
% \begin{equation*}
%     \begin{gathered}
        $\binom{n+d-2}{d-1} = \frac{nd}{(n+d)(n+d-1)}\binom{n+d}{d}$,
%     \end{gathered}
% \end{equation*}
we obtain
    \begin{align*}
        &\mathcal{O}\left(2\frac{n+d-1}{n}\binom{n+d-2}{d-1}^3 + 2\binom{n+d-2}{d-1}^2\right) = \mathcal{O}\left(2\frac{n+d-1}{n}\binom{n+d-2}{d-1}^3\right) \\
        &= \mathcal{O}\left(2\frac{n+d-1}{n}\frac{{(nd)}^3}{{(n+d)}^3{(n+d-1)}^3}\binom{n+d}{d}^3\right)\\
        &= \mathcal{O}\left(\frac{2n^2d^3}{{(n+d)}^3{(n+d-1)}^2}\binom{n+d}{d}^3\right)= \mathcal{O}\left(\binom{n+d}{d}^3\right),
    \end{align*}
    where the last equality holds, under the condition \(n > d\).
\end{proof}

\begin{remark}[On the number of $S$-polynomials]
The complexity estimate above assumes that exactly $\binom{n+i-1}{i}$ linearly independent $S$-polynomials are produced at the iteration-$i$ and that none of them reduces to zero.
This assumption is justified by the density of the intermediate systems:
at the beginning of the iteration-$i$, all monomials of degree at most $i+1$ are already present, and after completion the same property holds for degree at most $i$.
In particular, for a generic input system over $\Z_q$, the resulting $S$-polynomials are linearly independent with high probability (\Cref{pivots}).
If additional $S$-polynomials were required due to linear dependencies, their number would be bounded by a constant (or grow at most linearly) with respect to $\binom{n+i-1}{i}$.
Hence, the asymptotic complexity remains unchanged.
\end{remark}
